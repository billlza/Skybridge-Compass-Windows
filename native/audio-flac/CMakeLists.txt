cmake_minimum_required(VERSION 3.24)
project(SkyBridgeFlac VERSION 1.0.0 LANGUAGES C)
enable_testing()
# All native allocations are created and released inside this DLL. Carry its CRT
# statically so the optional decoder introduces no machine-wide runtime installer.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

# One unchanged shared decoder boundary; updates must refresh the reviewed source lock.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/source-lock.json" source_lock)
foreach(path SkyBridgeFLACDecoder.c include/SkyBridgeFLACDecoder.h LICENSE.SkyBridge COPYING.Xiph libflac-1.5.0-crc-fixture-alignment.patch)
    string(JSON expected GET "${source_lock}" bridgeFiles "${path}" sha256)
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${path}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Shared FLAC source differs from its lock: ${path}")
    endif()
endforeach()

include(FetchContent)
find_package(Git REQUIRED)
string(JSON flac_url GET "${source_lock}" libflac url)
string(JSON flac_sha GET "${source_lock}" libflac sha256)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_CXXLIBS OFF CACHE BOOL "" FORCE)
set(BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING ON CACHE BOOL "" FORCE)
set(WITH_OGG OFF CACHE BOOL "" FORCE)
# libFLAC 1.5's ENABLE_WERROR adds the literal GNU -Werror flag, even when
# its compiler probe rejected it. MSVC uses CMake's equivalent /WX property
# for every native target, including the upstream library and its tests.
if(MSVC)
    set(ENABLE_WERROR OFF CACHE BOOL "Use compiler-native warning errors" FORCE)
    set(CMAKE_COMPILE_WARNING_AS_ERROR ON)
else()
    set(ENABLE_WERROR ON CACHE BOOL "" FORCE)
endif()
set(WITH_STACK_PROTECTOR ON CACHE BOOL "" FORCE)
set(WITH_FORTIFY_SOURCE ON CACHE BOOL "" FORCE)
set(INSTALL_MANPAGES OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
FetchContent_Declare(flac URL "${flac_url}" URL_HASH "SHA256=${flac_sha}" DOWNLOAD_EXTRACT_TIMESTAMP TRUE
    PATCH_COMMAND "${GIT_EXECUTABLE}" apply --whitespace=error-all "${CMAKE_CURRENT_SOURCE_DIR}/libflac-1.5.0-crc-fixture-alignment.patch")
FetchContent_MakeAvailable(flac)
add_test(NAME skybridge_libFLAC COMMAND $<TARGET_FILE:test_libFLAC>)
set_tests_properties(skybridge_libFLAC PROPERTIES TIMEOUT 180 WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")

add_library(skybridge_flac SHARED SkyBridgeFLACDecoder.c SkyBridgeFLACABI.c)
target_include_directories(skybridge_flac PRIVATE include)
target_compile_definitions(skybridge_flac PRIVATE FLAC__NO_DLL)
target_link_libraries(skybridge_flac PRIVATE FLAC)
set_target_properties(skybridge_flac PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
if(MSVC)
    target_compile_options(skybridge_flac PRIVATE /W4 /WX /sdl /guard:cf)
    target_link_options(skybridge_flac PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
    target_sources(skybridge_flac PRIVATE SkyBridgeFLAC.def)
else()
    target_compile_options(skybridge_flac PRIVATE -Wall -Wextra -Werror -fstack-protector-strong)
    target_compile_definitions(skybridge_flac PRIVATE _FORTIFY_SOURCE=2)
endif()
