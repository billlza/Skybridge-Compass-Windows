cmake_minimum_required(VERSION 3.24)
project(SkyBridgeAlac VERSION 1.0.0 LANGUAGES C)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The qualified ALAC boundary requires a 64-bit target.")
endif()

# Consume a separately qualified provider build, never an ambient FFmpeg install.
# A local receipt binds bytes and configuration; it is not a release attestation.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/source-lock.json" lock)
string(JSON file_count LENGTH "${lock}" files)
math(EXPR file_last "${file_count} - 1")
foreach(i RANGE 0 ${file_last})
    string(JSON path MEMBER "${lock}" files ${i})
    string(JSON expected GET "${lock}" files "${path}")
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${path}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "ALAC boundary differs from its source lock: ${path}")
    endif()
endforeach()
if(NOT EXISTS "${SKYBRIDGE_ALAC_PROVIDER_RECEIPT}")
    message(FATAL_ERROR "Set SKYBRIDGE_ALAC_PROVIDER_RECEIPT to a qualified static provider receipt.")
endif()
file(READ "${SKYBRIDGE_ALAC_PROVIDER_RECEIPT}" provider)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/provider-source-lock.json" provider_lock)
foreach(key version archiveSha256 providerContract)
    string(JSON expected GET "${provider_lock}" "${key}")
    string(JSON actual GET "${provider}" "${key}")
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "ALAC provider receipt disagrees with its source lock: ${key}")
    endif()
endforeach()
string(JSON state GET "${provider}" status)
string(JSON baseline GET "${provider}" baseline)
string(JSON component_count LENGTH "${provider}" components)
string(JSON component GET "${provider}" components 0)
if(NOT state STREQUAL "built" OR baseline OR NOT component_count EQUAL 1 OR NOT component STREQUAL "CONFIG_ALAC_DECODER")
    message(FATAL_ERROR "ALAC requires the qualified minimal strict-error provider.")
endif()
if(APPLE)
    string(JSON slice GET "${provider}" appleSlice)
    if(NOT slice STREQUAL "macos-arm64" OR NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
        message(FATAL_ERROR "The available Apple provider must match an explicit macOS arm64 contract-host build.")
    endif()
elseif(WIN32)
    string(JSON target GET "${provider}" target)
    if(CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
        set(expected_target "windows-arm64")
    elseif(CMAKE_GENERATOR_PLATFORM STREQUAL "x64")
        set(expected_target "windows-x64")
    else()
        message(FATAL_ERROR "Select an explicit x64 or ARM64 Windows generator platform.")
    endif()
    if(NOT target STREQUAL expected_target)
        message(FATAL_ERROR "ALAC provider and Windows build target differ.")
    endif()
else()
    message(FATAL_ERROR "This boundary has no qualified provider profile for the selected platform.")
endif()

string(JSON source GET "${provider}" source)
string(JSON install GET "${provider}" install)
set(provider_patches patch buildPatch)
if(WIN32)
    list(APPEND provider_patches windowsBuildPatch)
endif()
set(provider_paths "")
foreach(kind IN LISTS provider_patches)
    string(JSON count LENGTH "${provider_lock}" ${kind} files)
    math(EXPR last "${count} - 1")
    foreach(i RANGE 0 ${last})
        string(JSON path GET "${provider_lock}" ${kind} files ${i} path)
        string(JSON expected GET "${provider_lock}" ${kind} files ${i} patchedSha256)
        # Later platform patches may refine the same build script. Validate the
        # final pinned bytes after composing the ordered patch chain.
        string(SHA256 key "${path}")
        set(provider_hash_${key} "${expected}")
        list(APPEND provider_paths "${path}")
    endforeach()
endforeach()
list(REMOVE_DUPLICATES provider_paths)
foreach(path IN LISTS provider_paths)
    string(SHA256 key "${path}")
    file(SHA256 "${source}/${path}" actual)
    if(NOT actual STREQUAL provider_hash_${key})
        message(FATAL_ERROR "ALAC provider patched source changed: ${path}")
    endif()
endforeach()
file(SHA256 "${install}/include/libavcodec/avcodec.h" installed_header)
file(SHA256 "${source}/libavcodec/avcodec.h" source_header)
if(NOT installed_header STREQUAL source_header)
    message(FATAL_ERROR "ALAC provider installed public header differs from patched source.")
endif()
string(JSON library_count LENGTH "${provider}" libraries)
if(NOT library_count EQUAL 2)
    message(FATAL_ERROR "ALAC requires exactly the qualified codec and utility static libraries.")
endif()
set(provider_libraries "")
foreach(i RANGE 0 1)
    string(JSON path GET "${provider}" libraries ${i} path)
    string(JSON expected GET "${provider}" libraries ${i} sha256)
    file(SHA256 "${path}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "ALAC provider library changed: ${path}")
    endif()
    list(APPEND provider_libraries "${path}")
endforeach()

add_library(skybridge_alac SHARED SkyBridgeALACDecoder.c SkyBridgeALACABI.c)
target_include_directories(skybridge_alac PRIVATE include "${install}/include")
target_link_libraries(skybridge_alac PRIVATE ${provider_libraries})
set_target_properties(skybridge_alac PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
if(MSVC)
    # SDK dependencies selected by the minimal Windows avutil configuration.
    # Static provider code is linked here; no FFmpeg/MSYS runtime DLL is admitted.
    target_link_libraries(skybridge_alac PRIVATE bcrypt user32)
    target_compile_options(skybridge_alac PRIVATE /W4 /WX /sdl /guard:cf)
    target_link_options(skybridge_alac PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
    target_sources(skybridge_alac PRIVATE SkyBridgeALAC.def)
elseif(APPLE)
    target_compile_options(skybridge_alac PRIVATE -Wall -Wextra -Werror -fstack-protector-strong)
    target_compile_definitions(skybridge_alac PRIVATE _FORTIFY_SOURCE=2)
    target_link_options(skybridge_alac PRIVATE "-Wl,-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/exports-macos.txt" -Wl,-dead_strip -Wl,-fatal_warnings)
else()
    message(FATAL_ERROR "Windows provider qualification requires the MSVC toolchain.")
endif()
